cmake_minimum_required(VERSION 3.24)
project(TrieFrameViewer VERSION 1.1.3 LANGUAGES CXX)

if(NOT WIN32)
    message(FATAL_ERROR "Trie FrameViewer currently targets Windows only.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_INSTALL_BINDIR "." CACHE STRING "Portable executable directory")
set(CMAKE_INSTALL_LIBDIR "." CACHE STRING "Portable runtime library directory")
if(POLICY CMP0144)
    cmake_policy(SET CMP0144 NEW)
endif()
include(CTest)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Multimedia ShaderTools)
if(BUILD_TESTING)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
endif()
find_package(FFmpeg REQUIRED)
qt_standard_project_setup(REQUIRES 6.8)
add_subdirectory(src)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
if(WIN32)
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
    include(InstallRequiredSystemLibraries)
endif()
